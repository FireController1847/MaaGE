# Root integration hook (Agent 11): add_subdirectory(src/audio/core)
# The target is asset-free and independent of World, rendering and gameplay.
add_library(maage_audio STATIC audio.c mixer.c audio_lock.c)
add_library(MaaGE::Audio ALIAS maage_audio)
maage_configure_portable_target(maage_audio)
target_include_directories(maage_audio PUBLIC "${PROJECT_SOURCE_DIR}/include")
if(NOT TARGET MaaGE::Thread)
    add_subdirectory("${PROJECT_SOURCE_DIR}/src/core/thread" "${PROJECT_BINARY_DIR}/audio-thread")
endif()
target_link_libraries(maage_audio PUBLIC MaaGE::Status MaaGE::Thread)
if(NOT MAAGE_PLATFORM_WINDOWS)
    target_link_libraries(maage_audio PRIVATE m)
endif()
if(MAAGE_PLATFORM_MACOS)
    target_sources(maage_audio PRIVATE backend_coreaudio.c)
    target_link_libraries(maage_audio PRIVATE "-framework AudioToolbox" "-framework AudioUnit")
elseif(MAAGE_PLATFORM_WINDOWS)
    target_sources(maage_audio PRIVATE backend_wasapi.c)
    target_link_libraries(maage_audio PRIVATE ole32 uuid)
elseif(MAAGE_PLATFORM_LINUX)
    target_sources(maage_audio PRIVATE backend_alsa.c)
    target_link_libraries(maage_audio PRIVATE ${CMAKE_DL_LIBS})
else()
    target_sources(maage_audio PRIVATE backend_stub.c)
endif()

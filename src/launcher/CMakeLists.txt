if(NOT MAAGE_BUILD_MINECRAFT_ASSETS)
    message(FATAL_ERROR "Launcher setup requires MAAGE_BUILD_MINECRAFT_ASSETS=ON.")
endif()
if(NOT MAAGE_BUILD_INSTANCE)
    message(FATAL_ERROR "Launcher requires MAAGE_BUILD_INSTANCE=ON.")
endif()
find_package(json-c CONFIG REQUIRED)
find_package(Threads REQUIRED)
if(MAAGE_PLATFORM_LINUX)
    # Distro CMake exports can require uninstalled libzip command-line tools.
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(MAAGE_LIBZIP REQUIRED IMPORTED_TARGET libzip>=1.7)
    set(MAAGE_LIBZIP_TARGET PkgConfig::MAAGE_LIBZIP)
else()
    find_package(libzip 1.7 CONFIG REQUIRED)
    set(MAAGE_LIBZIP_TARGET libzip::zip)
endif()
find_package(CURL 7.85 REQUIRED)
find_package(OpenSSL 1.1.0 REQUIRED COMPONENTS Crypto)
add_library(maage_launcher_access STATIC access/access.c access/accounts.c access/verify.c access/profile_store.c access/oauth.c access/auth_http.c access/auth_transport.c)
maage_configure_portable_target(maage_launcher_access)
target_link_libraries(maage_launcher_access PRIVATE MaaGE::Storage maage_core CURL::libcurl OpenSSL::Crypto json-c::json-c Threads::Threads)
add_library(maage_launcher_assets STATIC assets.c client_resources.c)
maage_configure_portable_target(maage_launcher_assets)
target_link_libraries(maage_launcher_assets PUBLIC MaaGE::Download maage_launcher_access PRIVATE MaaGE::MinecraftAssets MaaGE::StorageHash json-c::json-c Threads::Threads ${MAAGE_LIBZIP_TARGET} OpenSSL::Crypto)
if(MAAGE_PLATFORM_LINUX)
    target_link_libraries(maage_launcher_assets PRIVATE m)
endif()
option(MAAGE_LAUNCHER_DEVELOPER_IMPORT "Enable trusted local Client import outside Debug builds" OFF)
maage_client_target_architecture(MAAGE_CLIENT_HOST_ARCH)
add_library(maage_launcher_clients STATIC clients/registry.c clients/files.c clients/semver.c clients/job.c clients/launch.c)
maage_configure_portable_target(maage_launcher_clients)
target_compile_definitions(maage_launcher_clients PUBLIC
    "MAAGE_LAUNCHER_DEVELOPMENT=$<OR:$<CONFIG:Debug>,$<BOOL:${MAAGE_LAUNCHER_DEVELOPER_IMPORT}>>")
target_compile_definitions(maage_launcher_clients PRIVATE
    "MAAGE_CLIENT_HOST_PLATFORM=\"${MAAGE_TARGET_SYSTEM}\""
    "MAAGE_CLIENT_HOST_ARCH=\"${MAAGE_CLIENT_HOST_ARCH}\"")
target_link_libraries(maage_launcher_clients PUBLIC MaaGE::Storage MaaGE::Process MaaGE::Instance PRIVATE maage_launcher_access json-c::json-c OpenSSL::Crypto Threads::Threads)
add_library(maage_launcher_ui STATIC ui/launcher.c)
maage_configure_portable_target(maage_launcher_ui)
target_link_libraries(maage_launcher_ui PUBLIC maage_launcher_assets maage_launcher_clients PRIVATE MaaGE::Storage json-c::json-c maage_core)
add_executable(maage_launcher main.c demonstration.c native/setup.c)
if(MAAGE_PLATFORM_MACOS)
    enable_language(OBJC)
    target_sources(maage_launcher_access PRIVATE native/auth_macos.m)
    set_source_files_properties(native/auth_macos.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc;-std=c99")
    target_link_libraries(maage_launcher_access PRIVATE "-framework Security" "-framework CoreServices")
    target_sources(maage_launcher PRIVATE native/setup_macos.m native/access_macos.m ui/launcher_macos.m)
    set_source_files_properties(native/setup_macos.m native/access_macos.m ui/launcher_macos.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc;-std=c99")
    target_link_libraries(maage_launcher PRIVATE "-framework AppKit" "-framework QuartzCore")
elseif(MAAGE_PLATFORM_WINDOWS)
    target_sources(maage_launcher_access PRIVATE native/auth_win32.c)
    target_link_libraries(maage_launcher_access PRIVATE bcrypt ws2_32 shell32)
    target_sources(maage_launcher PRIVATE native/setup_win32.c native/access_win32.c ui/launcher_win32.c)
    target_link_libraries(maage_launcher PRIVATE comctl32 comdlg32 uxtheme shell32 ole32)
elseif(MAAGE_PLATFORM_LINUX)
    pkg_check_modules(MAAGE_GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.22)
    add_library(maage_launcher_auth_native STATIC native/auth_linux.c)
    maage_configure_portable_target(maage_launcher_auth_native)
    set_target_properties(maage_launcher_auth_native PROPERTIES C_STANDARD 99)
    target_link_libraries(maage_launcher_auth_native PRIVATE PkgConfig::MAAGE_GTK MaaGE::Log)
    target_link_libraries(maage_launcher_access PRIVATE maage_launcher_auth_native)
    add_library(maage_launcher_native STATIC native/setup_gtk.c native/access_gtk.c ui/launcher_gtk.c)
    maage_configure_portable_target(maage_launcher_native)
    # GTK system headers require modern C; this exception stays in the adapter.
    set_target_properties(maage_launcher_native PROPERTIES C_STANDARD 99)
    add_dependencies(maage_launcher_native maage_versions)
    target_link_libraries(maage_launcher_native PRIVATE PkgConfig::MAAGE_GTK maage_launcher_ui)
    target_link_libraries(maage_launcher PRIVATE maage_launcher_native)
else()
    message(FATAL_ERROR "Launcher setup requires macOS, Windows, or Linux.")
endif()
add_dependencies(maage_launcher maage_versions)
maage_configure_portable_target(maage_launcher)
set_target_properties(maage_launcher PROPERTIES OUTPUT_NAME "${MAAGE_LAUNCHER_NAME}")
maage_configure_gui_application(maage_launcher launcher)
target_compile_definitions(maage_launcher PRIVATE MAAGE_APPLICATION_USER_DATA)
target_link_libraries(maage_launcher PRIVATE MaaGE::Render maage_launcher_ui)
if(MAAGE_PLATFORM_LINUX)
    target_link_libraries(maage_launcher PRIVATE m)
endif()

if(MSVC)
    target_compile_options(maage_launcher PRIVATE /utf-8)
endif()

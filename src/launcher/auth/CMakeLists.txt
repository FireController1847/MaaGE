find_package(CURL 7.85 REQUIRED)
find_package(json-c CONFIG REQUIRED)
find_package(Threads REQUIRED)

set(MAAGE_MICROSOFT_CLIENT_ID "" CACHE STRING "MaaGE's own registered Microsoft public-client application ID")
set(MAAGE_AUTH_AUTHORIZE_URL "https://login.microsoftonline.com/consumers/oauth2/v2.0/authorize" CACHE STRING "Microsoft authorization endpoint")
set(MAAGE_AUTH_TOKEN_URL "https://login.microsoftonline.com/consumers/oauth2/v2.0/token" CACHE STRING "Microsoft token endpoint")
set(MAAGE_AUTH_XBOX_URL "https://user.auth.xboxlive.com/user/authenticate" CACHE STRING "Xbox user authentication endpoint")
set(MAAGE_AUTH_XSTS_URL "https://xsts.auth.xboxlive.com/xsts/authorize" CACHE STRING "Xbox XSTS endpoint")
set(MAAGE_AUTH_MINECRAFT_URL "https://api.minecraftservices.com" CACHE STRING "Minecraft Services base URL")
set(MAAGE_AUTH_CONFIG "/* Generated by CMake. No client secret belongs here. */\n")
foreach(name MICROSOFT_CLIENT_ID AUTH_AUTHORIZE_URL AUTH_TOKEN_URL AUTH_XBOX_URL AUTH_XSTS_URL AUTH_MINECRAFT_URL)
    if(MAAGE_${name} MATCHES "[\"\\\n\r]")
        message(FATAL_ERROR "Invalid authentication configuration: MAAGE_${name}")
    endif()
    if(NOT name STREQUAL "MICROSOFT_CLIENT_ID" AND NOT MAAGE_${name} MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?(/[^ ?#@]*)?$")
        message(FATAL_ERROR "Authentication endpoints must be absolute HTTPS URLs without credentials, queries or fragments.")
    endif()
    string(APPEND MAAGE_AUTH_CONFIG "#define MAAGE_${name} \"${MAAGE_${name}}\"\n")
endforeach()
file(CONFIGURE OUTPUT "${PROJECT_BINARY_DIR}/generated/maage/auth_config.h" CONTENT "${MAAGE_AUTH_CONFIG}" @ONLY)

add_library(maage_auth_adapters STATIC http_json.c platform.c metadata.c loopback.c avatar.c)
# Dependency/OS headers require modern C; keep this exception private to adapters.
set_target_properties(maage_auth_adapters PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(maage_auth_adapters PRIVATE "${PROJECT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(maage_auth_adapters PRIVATE CURL::libcurl json-c::json-c Threads::Threads)
if(MAAGE_PLATFORM_WINDOWS)
    target_compile_definitions(maage_auth_adapters PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(maage_auth_adapters PRIVATE bcrypt advapi32 shell32 ws2_32)
elseif(MAAGE_PLATFORM_MACOS)
    find_library(MAAGE_AUTH_SECURITY Security REQUIRED)
    find_library(MAAGE_AUTH_SERVICES CoreServices REQUIRED)
    find_library(MAAGE_AUTH_IMAGES ImageIO REQUIRED)
    find_library(MAAGE_AUTH_GRAPHICS CoreGraphics REQUIRED)
    target_link_libraries(maage_auth_adapters PRIVATE "${MAAGE_AUTH_SECURITY}" "${MAAGE_AUTH_SERVICES}" "${MAAGE_AUTH_IMAGES}" "${MAAGE_AUTH_GRAPHICS}")
elseif(MAAGE_PLATFORM_LINUX)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(MAAGE_AUTH_SECRET REQUIRED IMPORTED_TARGET libsecret-1)
    target_link_libraries(maage_auth_adapters PRIVATE PkgConfig::MAAGE_AUTH_SECRET)
else()
    message(FATAL_ERROR "No native secure-storage adapter for ${CMAKE_SYSTEM_NAME}")
endif()
add_library(maage_launcher_auth STATIC accounts.c config.c oauth.c xbox.c minecraft.c store.c)
add_library(MaaGE::LauncherAuth ALIAS maage_launcher_auth)
maage_configure_portable_target(maage_launcher_auth)
target_include_directories(maage_launcher_auth PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(maage_launcher_auth PRIVATE maage_auth_adapters maage_core PUBLIC MaaGE::LauncherState)

add_library(maage_canvas STATIC)
add_library(MaaGE::Canvas ALIAS maage_canvas)
maage_configure_portable_target(maage_canvas)
target_include_directories(maage_canvas PUBLIC "${PROJECT_SOURCE_DIR}/include")
if(MAAGE_PLATFORM_MACOS)
    enable_language(OBJC)
    target_sources(maage_canvas PRIVATE canvas_macos.m)
    find_library(MAAGE_COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
    find_library(MAAGE_CORETEXT_FRAMEWORK CoreText REQUIRED)
    target_link_libraries(maage_canvas PRIVATE "${MAAGE_COREGRAPHICS_FRAMEWORK}" "${MAAGE_CORETEXT_FRAMEWORK}")
else()
    target_sources(maage_canvas PRIVATE canvas_unavailable.c)
endif()

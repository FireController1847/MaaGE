add_library(maage_crypto STATIC crypto.c)
add_library(MaaGE::Crypto ALIAS maage_crypto)
maage_configure_portable_target(maage_crypto)
target_include_directories(maage_crypto PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(maage_crypto PUBLIC MaaGE::Status)
if(MAAGE_PLATFORM_WINDOWS)
    target_link_libraries(maage_crypto PRIVATE bcrypt)
else()
    find_package(OpenSSL 1.1.0 REQUIRED COMPONENTS Crypto)
    target_link_libraries(maage_crypto PRIVATE OpenSSL::Crypto)
endif()

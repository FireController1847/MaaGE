add_library(maage_window STATIC window.c)
add_library(MaaGE::Window ALIAS maage_window)
maage_configure_portable_target(maage_window)
target_include_directories(maage_window PUBLIC "${PROJECT_SOURCE_DIR}/include")

if(MAAGE_PLATFORM_WINDOWS)
    target_sources(maage_window PRIVATE win32/window_win32.c)
    target_compile_definitions(maage_window PRIVATE MAAGE_WINDOW_HAS_WIN32)
    target_link_libraries(maage_window PRIVATE user32 gdi32)
elseif(MAAGE_PLATFORM_MACOS)
    enable_language(OBJC)
    find_library(MAAGE_APPKIT_FRAMEWORK AppKit REQUIRED)
    find_library(MAAGE_COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
    target_sources(maage_window PRIVATE macos/window_macos.m)
    target_compile_definitions(maage_window PRIVATE MAAGE_WINDOW_HAS_COCOA)
    target_compile_options(maage_window PRIVATE "$<$<COMPILE_LANGUAGE:OBJC>:-fno-objc-arc>")
    target_link_libraries(maage_window PRIVATE "${MAAGE_APPKIT_FRAMEWORK}" "${MAAGE_COREGRAPHICS_FRAMEWORK}")
elseif(MAAGE_PLATFORM_LINUX)
    option(MAAGE_WINDOW_ENABLE_X11 "Build the native X11 adapter" ON)
    option(MAAGE_WINDOW_ENABLE_WAYLAND "Build the native Wayland adapter" ON)
    if(NOT MAAGE_WINDOW_ENABLE_X11 AND NOT MAAGE_WINDOW_ENABLE_WAYLAND)
        message(FATAL_ERROR "Native Linux windowing requires X11 and/or Wayland.")
    endif()
    if(MAAGE_WINDOW_ENABLE_X11)
        find_package(X11 REQUIRED)
        target_sources(maage_window PRIVATE linux/x11/window_x11.c)
        target_compile_definitions(maage_window PRIVATE MAAGE_WINDOW_HAS_X11)
        target_link_libraries(maage_window PRIVATE X11::X11)
    endif()
    if(MAAGE_WINDOW_ENABLE_WAYLAND)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(MAAGE_WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
        pkg_check_modules(MAAGE_WAYLAND_PROTOCOLS REQUIRED wayland-protocols)
        pkg_get_variable(MAAGE_WAYLAND_PROTOCOL_DIR wayland-protocols pkgdatadir)
        find_program(MAAGE_WAYLAND_SCANNER wayland-scanner REQUIRED)
        set(MAAGE_WAYLAND_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
        file(MAKE_DIRECTORY "${MAAGE_WAYLAND_GENERATED}")
        set(MAAGE_WAYLAND_PROTOCOL_SOURCES)
        foreach(protocol xdg-shell xdg-decoration)
            if(protocol STREQUAL "xdg-shell")
                set(MAAGE_PROTOCOL_XML "${MAAGE_WAYLAND_PROTOCOL_DIR}/stable/xdg-shell/xdg-shell.xml")
            else()
                set(MAAGE_PROTOCOL_XML "${MAAGE_WAYLAND_PROTOCOL_DIR}/unstable/xdg-decoration/xdg-decoration-unstable-v1.xml")
            endif()
            set(MAAGE_PROTOCOL_HEADER "${MAAGE_WAYLAND_GENERATED}/${protocol}-client-protocol.h")
            set(MAAGE_PROTOCOL_SOURCE "${MAAGE_WAYLAND_GENERATED}/${protocol}-protocol.c")
            add_custom_command(OUTPUT "${MAAGE_PROTOCOL_HEADER}" "${MAAGE_PROTOCOL_SOURCE}"
                COMMAND "${MAAGE_WAYLAND_SCANNER}" client-header "${MAAGE_PROTOCOL_XML}" "${MAAGE_PROTOCOL_HEADER}"
                COMMAND "${MAAGE_WAYLAND_SCANNER}" private-code "${MAAGE_PROTOCOL_XML}" "${MAAGE_PROTOCOL_SOURCE}"
                DEPENDS "${MAAGE_PROTOCOL_XML}" VERBATIM)
            list(APPEND MAAGE_WAYLAND_PROTOCOL_SOURCES "${MAAGE_PROTOCOL_HEADER}" "${MAAGE_PROTOCOL_SOURCE}")
        endforeach()
        # Generated Wayland inline helpers require C99. Do not weaken portable C.
        add_library(maage_window_wayland OBJECT linux/wayland/window_wayland.c ${MAAGE_WAYLAND_PROTOCOL_SOURCES})
        set_target_properties(maage_window_wayland PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
        target_include_directories(maage_window_wayland PRIVATE "${MAAGE_WAYLAND_GENERATED}" "${PROJECT_SOURCE_DIR}/include")
        target_link_libraries(maage_window_wayland PRIVATE PkgConfig::MAAGE_WAYLAND)
        target_sources(maage_window PRIVATE $<TARGET_OBJECTS:maage_window_wayland>)
        target_compile_definitions(maage_window PRIVATE MAAGE_WINDOW_HAS_WAYLAND)
        target_link_libraries(maage_window PRIVATE PkgConfig::MAAGE_WAYLAND)
    endif()
else()
    message(FATAL_ERROR "No native windowing adapter for ${CMAKE_SYSTEM_NAME}; disable MAAGE_BUILD_WINDOW for headless builds.")
endif()

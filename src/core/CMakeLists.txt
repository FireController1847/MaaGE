set(MAAGE_BUILD_GRAPHICAL_DEFAULT OFF)
if(MAAGE_BUILD_LAUNCHER OR MAAGE_BUILD_CLIENT)
    set(MAAGE_BUILD_GRAPHICAL_DEFAULT ON)
endif()
add_library(maage_core STATIC core.c time.c)
maage_configure_portable_target(maage_core)
target_include_directories(maage_core PUBLIC "${PROJECT_SOURCE_DIR}/include")
add_library(maage_status STATIC status.c)
add_library(MaaGE::Status ALIAS maage_status)
maage_configure_portable_target(maage_status)
target_include_directories(maage_status PUBLIC "${PROJECT_SOURCE_DIR}/include")
add_library(maage_geometry STATIC geometry.c)
add_library(MaaGE::Geometry ALIAS maage_geometry)
maage_configure_portable_target(maage_geometry)
target_include_directories(maage_geometry PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(maage_geometry PUBLIC MaaGE::Status)
add_subdirectory(compression)
add_subdirectory(log)
target_link_libraries(maage_core PUBLIC MaaGE::Log)

option(MAAGE_BUILD_WINDOW "Build the native Core windowing library" ${MAAGE_BUILD_GRAPHICAL_DEFAULT})
if(MAAGE_BUILD_WINDOW)
    add_subdirectory(window)
elseif(MAAGE_BUILD_LAUNCHER)
    message(FATAL_ERROR "The Launcher requires MAAGE_BUILD_WINDOW=ON.")
endif()

option(MAAGE_BUILD_RENDER "Build the native Core rendering library" ${MAAGE_BUILD_GRAPHICAL_DEFAULT})
if(MAAGE_BUILD_RENDER)
    if(NOT MAAGE_BUILD_WINDOW)
        message(FATAL_ERROR "MAAGE_BUILD_RENDER requires MAAGE_BUILD_WINDOW=ON.")
    endif()
    add_subdirectory(render)
elseif(MAAGE_BUILD_LAUNCHER)
    message(FATAL_ERROR "The Launcher requires MAAGE_BUILD_RENDER=ON.")
endif()

if(MAAGE_BUILD_LAUNCHER OR MAAGE_BUILD_MINECRAFT_ASSETS OR MAAGE_BUILD_IMAGE OR MAAGE_BUILD_WORLD OR MAAGE_BUILD_INSTANCE)
    if(MAAGE_BUILD_LAUNCHER)
        add_subdirectory(crypto)
    endif()
    add_subdirectory(storage)
endif()
option(MAAGE_BUILD_JOBS "Build the reusable bounded CPU job scheduler" ${MAAGE_BUILD_WORLD})
if((MAAGE_BUILD_LAUNCHER OR MAAGE_BUILD_WORLD OR MAAGE_BUILD_JOBS) AND NOT TARGET MaaGE::Thread)
    add_subdirectory(thread)
endif()
if(MAAGE_BUILD_JOBS)
    add_subdirectory(jobs)
endif()
if(MAAGE_BUILD_WORLD AND NOT MAAGE_BUILD_JOBS)
    message(FATAL_ERROR "MAAGE_BUILD_WORLD requires MAAGE_BUILD_JOBS=ON.")
endif()
if(MAAGE_BUILD_LAUNCHER)
    add_subdirectory(net)
    add_subdirectory(process)
endif()
